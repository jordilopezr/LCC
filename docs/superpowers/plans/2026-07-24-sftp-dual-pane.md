# SFTP Dual-Pane Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Turn the SFTP tab into a WinSCP-style dual-pane (local/remote) file manager with drag-and-drop transfers and a sequential transfer queue that supports real per-item and mid-transfer cancellation.

**Architecture:** Reuse the existing `SftpBrowserBody` as the remote pane unchanged; add a `LocalPane` (dart:io) and a `SftpDualView` container with a classic toggle. A per-tab `TransferQueueNotifier` runs one file at a time through the existing parallel engine. Mid-transfer cancel is new backend: a Rust cancellation-token registry keyed by a `u32` transfer id, flipped by a new `sftp_cancel_transfer` bridge call the worker threads already poll.

**Tech Stack:** Rust (ssh2, std::sync::OnceLock), flutter_rust_bridge 2.11.1, Flutter, Riverpod 3.x Notifier, gen-l10n (EN/ES parity).

```rust
/// Process-wide registry of in-flight transfer cancellation flags, keyed by the
/// Dart-generated transfer id. A running parallel transfer registers its flag
/// on start and removes it on finish; `sftp_cancel_transfer` flips it so the
/// worker threads (which already poll their `cancel` flag) abort and clean up.
fn cancellations() -> &'static Mutex<HashMap<u32, Arc<AtomicBool>>> {
    static C: OnceLock<Mutex<HashMap<u32, Arc<AtomicBool>>>> = OnceLock::new();
    C.get_or_init(|| Mutex::new(HashMap::new()))
}

fn register_cancel(id: u32) -> Arc<AtomicBool> {
    let flag = Arc::new(AtomicBool::new(false));
    cancellations().lock().unwrap().insert(id, Arc::clone(&flag));
    flag
}

fn unregister_cancel(id: u32) {
    cancellations().lock().unwrap().remove(&id);
}

/// Flip the cancellation flag for an in-flight transfer. No-op if the id is
/// unknown (already finished or never started).
pub fn sftp_cancel_transfer(transfer_id: u32) {
    if let Some(flag) = cancellations().lock().unwrap().get(&transfer_id) {
        flag.store(true, Ordering::Relaxed);
    }
}

/// Removes a transfer's cancel flag on scope exit, so every early return in the
/// parallel transfer paths unregisters exactly once.
struct CancelGuard(u32);
impl Drop for CancelGuard {
    fn drop(&mut self) {
        unregister_cancel(self.0);
    }
}
```

```dart
enum TransferDirection { upload, download }
enum TransferStatus { queued, active, done, failed, cancelled }

@immutable
class TransferItem {

  final String localPath;
  final String remotePath;
  final TransferDirection direction;
  final int sizeBytes;

        error: error ?? this.error,

}

@immutable
class TransferQueueState {
  final List<TransferItem> items;
  const TransferQueueState({this.items = const []});
  TransferQueueState copyWith({List<TransferItem>? items}) =>
      TransferQueueState(items: items ?? this.items);
}

/// Per-SftpTab, in-memory, sequential transfer queue. Overridable seams
/// (`runTransfer`, `cancelTransfer`) let tests inject fakes without the bridge.
class TransferQueueNotifier extends Notifier<TransferQueueState> {
  String host = '';
  int port = 0;
  String username = '';
  bool _running = false;
  int? _activeId;

  @override
  TransferQueueState build() => const TransferQueueState();

  @protected
  Stream<SftpProgress> runTransfer(TransferItem item) {
    switch (item.direction) {
      case TransferDirection.upload:
        return sftpUploadParallel(
          host: host, port: port, username: username,
          localPath: item.localPath, remotePath: item.remotePath,
          transferId: item.id,
        );
      case TransferDirection.download:
        return sftpDownloadParallel(
          host: host, port: port, username: username,
          remotePath: item.remotePath, localPath: item.localPath,
          transferId: item.id,
        );
    }
  }

  @protected
  Future<void> cancelTransfer(int transferId) =>
      sftpCancelTransfer(transferId: transferId);

  void _update(int id, TransferItem Function(TransferItem) f) {
    state = state.copyWith(items: [
      for (final i in state.items) if (i.id == id) f(i) else i,
    ]);
  }

  void enqueue(List<TransferItem> items) {
    state = state.copyWith(items: [...state.items, ...items]);
    _pump();
  }

  void cancel(int id) {
    final item = state.items.firstWhere((i) => i.id == id,
        orElse: () => throw StateError('no such transfer'));
    if (item.status == TransferStatus.queued) {
      _update(id, (i) => i.copyWith(status: TransferStatus.cancelled));
    } else if (item.status == TransferStatus.active) {
      // Mark cancelled up-front (deterministic), then flip the Rust token; the
      // running stream errors out and `_runOne`'s catch leaves it cancelled.
      _update(id, (i) => i.copyWith(status: TransferStatus.cancelled));
      cancelTransfer(id);
    }
  }

  void retry(int id) {
    _update(id, (i) => i.copyWith(status: TransferStatus.queued, transferred: 0, error: null));
    _pump();
  }

  void clearCompleted() {
    state = state.copyWith(
        items: state.items.where((i) => i.status != TransferStatus.done).toList());
  }

  bool get hasActiveTransfers => state.items.any(
      (i) => i.status == TransferStatus.queued || i.status == TransferStatus.active);

  Future<void> _pump() async {
    if (_running) return;
    _running = true;
    try {
      while (true) {
        final next = state.items
            .where((i) => i.status == TransferStatus.queued)
            .cast<TransferItem?>()
            .firstWhere((i) => true, orElse: () => null);
        if (next == null) break;
        await _runOne(next);
      }
    } finally {
      _running = false;
    }
  }

  Future<void> _runOne(TransferItem item) async {
    _activeId = item.id;
    _update(item.id, (i) => i.copyWith(status: TransferStatus.active));
    try {
      await for (final p in runTransfer(item)) {
        _update(item.id, (i) => i.copyWith(transferred: p.transferred.toInt()));
      }
      _update(item.id, (i) => i.copyWith(
          status: TransferStatus.done, transferred: i.sizeBytes));
    } catch (e) {
      // If the user cancelled, `cancel()` already set the status to cancelled;
      // leave it. Otherwise it's a real failure.
      final cur = state.items.firstWhere((i) => i.id == item.id);
      if (cur.status != TransferStatus.cancelled) {
        _update(item.id, (i) => i.copyWith(status: TransferStatus.failed, error: e.toString()));
      }
    } finally {
      _activeId = null;
    }
  }
}

NotifierProvider<TransferQueueNotifier, TransferQueueState> createTransferQueueProvider({
  required String host,
  required int port,
  required String username,
}) {
  return NotifierProvider<TransferQueueNotifier, TransferQueueState>(() {
    final n = TransferQueueNotifier();
    n.host = host;
    n.port = port;
    n.username = username;
    return n;
  });
}
```

```dart
import 'dart:io';
import 'package:flutter/material.dart';
import 'package:path/path.dart' as p;

class LocalEntry {
  final String name;
  final String path;
  final bool isDirectory;
  final int sizeBytes;
  const LocalEntry({
    required this.name,
    required this.path,
    required this.isDirectory,
    required this.sizeBytes,
  });
}

/// Lists a local directory: directories first, then files, each group sorted
/// case-insensitively by name. Symlinks/other are treated as files.
Future<List<LocalEntry>> listLocalDir(String path) async {
  final dir = Directory(path);
  final out = <LocalEntry>[];
  await for (final e in dir.list(followLinks: false)) {
    final stat = await e.stat();
    final isDir = stat.type == FileSystemEntityType.directory;
    out.add(LocalEntry(
      name: p.basename(e.path),
      path: e.path,
      isDirectory: isDir,
      sizeBytes: isDir ? 0 : stat.size,
    ));
  }
  out.sort((a, b) {
    if (a.isDirectory != b.isDirectory) return a.isDirectory ? -1 : 1;
    return a.name.toLowerCase().compareTo(b.name.toLowerCase());
  });
  return out;
}

class LocalPane extends StatefulWidget {
  final void Function(String path)? onDirChanged;
  const LocalPane({super.key, this.onDirChanged});

  @override
  State<LocalPane> createState() => LocalPaneState();
}

class LocalPaneState extends State<LocalPane> {
  late String _cwd;
  List<LocalEntry> _entries = [];
  String? _error;

  @override
  void initState() {
    super.initState();
    _cwd = Platform.environment['HOME'] ?? Directory.current.path;
    _load();
  }

  String get currentDir => _cwd;

  Future<void> _load() async {
    try {
      final entries = await listLocalDir(_cwd);
      if (!mounted) return;
      setState(() {
        _entries = entries;
        _error = null;
      });
      widget.onDirChanged?.call(_cwd);
    } catch (e) {
      if (!mounted) return;
      setState(() => _error = e.toString());
    }
  }

  void _enter(LocalEntry e) {
    if (!e.isDirectory) return;
    setState(() => _cwd = e.path);
    _load();
  }

  void _up() {
    final parent = p.dirname(_cwd);
    if (parent == _cwd) return;
    setState(() => _cwd = parent);
    _load();
  }

  @override
  Widget build(BuildContext context) {
    final scheme = Theme.of(context).colorScheme;
    return Column(children: [
      Container(
        color: scheme.surfaceContainerHighest,
        padding: const EdgeInsets.symmetric(horizontal: 8, vertical: 4),
        child: Row(children: [
          const Icon(Icons.computer, size: 16),
          const SizedBox(width: 6),
          Expanded(child: Text(_cwd, overflow: TextOverflow.ellipsis)),
        ]),
      ),
      const Divider(height: 1),
      Expanded(
        child: _error != null
            ? Center(child: Text(_error!))
            : ListView(children: [
                ListTile(
                  dense: true,
                  leading: const Icon(Icons.arrow_upward, size: 18),
                  title: const Text('..'),
                  onTap: _up,
                ),
                for (final e in _entries)
                  ListTile(
                    key: ValueKey('local-${e.path}'),
                    dense: true,
                    leading: Icon(
                        e.isDirectory ? Icons.folder : Icons.insert_drive_file,
                        size: 18),
                    title: Text(e.name),
                    onTap: () => _enter(e),
                  ),
              ]),
      ),
    ]);
  }
}
```

```dart
import 'transfer_id.dart';
import 'transfer_queue_provider.dart';

enum ConflictChoice { overwriteAll, skipExisting, cancel }

/// One file to transfer, with its path relative to the dragged root (so folders
/// expand to their files while preserving subdirectory structure).
class SourceFile {
  final String path;         // absolute source path (local for upload, remote for download)
  final String relativePath; // path relative to the drop, e.g. "sub/b.txt"
  final int sizeBytes;
  const SourceFile({required this.path, required this.relativePath, required this.sizeBytes});

  String get topName => relativePath.split('/').first;
}

/// Turns a resolved drop (files already enumerated, destination listing known)
/// into queue items, applying the batch conflict choice. Pure + testable.
List<TransferItem> buildTransferItems({
  required List<SourceFile> files,
  required TransferDirection direction,
  required String destDir,
  required Set<String> existingNames,
  required ConflictChoice choice,
}) {
  if (choice == ConflictChoice.cancel) return const [];
  final out = <TransferItem>[];
  for (final f in files) {
    if (choice == ConflictChoice.skipExisting && existingNames.contains(f.topName)) {
      continue;
    }
    final dest = '$destDir/${f.relativePath}';
    out.add(TransferItem(
      id: nextTransferId(),
      localPath: direction == TransferDirection.upload ? f.path : dest,
      remotePath: direction == TransferDirection.upload ? dest : f.path,
      direction: direction,
      sizeBytes: f.sizeBytes,
      status: TransferStatus.queued,
      transferred: 0,
    ));
  }
  return out;
}
```
